% 1. EXECUTIVE SUMMARY
% ===================================================================
\mkmexecsummary

This document presents the mathematical framework and implementation details for applying
Uniform Manifold Approximation and Projection (UMAP) to weather-to-flood modelling. The
approach treats high-dimensional atmospheric data as residing on a Riemannian manifold,
enabling dimensionality reduction that preserves the essential topological structures
governing fluid dynamics over terrain.

The model reduces computational cost by approximately 33$\times$ relative to full CFD
simulation while maintaining flood prediction RMSE within 0.31\,m (compared to 0.24\,m
for full physics). It achieves 87.3\% accuracy in unsupervised storm classification and
42\% lower topological distortion than linear methods (PCA).

\textbf{Key risk:} The model is used as an upstream component feeding the hazard curve
and PRS pricing models. Errors in dimensionality reduction propagate to flood probability
estimates and, consequently, to swap pricing.

% 2. MODEL PURPOSE AND SCOPE

\mkmpurpose

\subsection{Purpose}
UMAP is applied to reduce the dimensionality of meteorological forecast fields before
they enter the hydrodynamic flood model. This enables:
\begin{itemize}[nosep]
    \item Real-time processing of HRRR 3\,km forecast data within operational latency
          constraints ($<$10 minutes)
    \item Preservation of multi-scale atmospheric structures (synoptic, mesoscale,
          microscale) that drive flood response
    \item Uncertainty quantification through ensemble processing in reduced dimensions
\end{itemize}

\subsection{Scope}
\begin{itemize}[nosep]
    \item \textbf{In scope:} Dimensionality reduction of atmospheric fields; mapping to
          hydrodynamic initial conditions; topology preservation metrics
    \item \textbf{Out of scope:} The hydrodynamic solver itself (shallow water equations);
          property-level damage estimation; PRS pricing
    \item \textbf{Upstream dependencies:} HRRR forecast data (3\,km, hourly)
    \item \textbf{Downstream consumers:} Gauge response model, hazard curve builder,
          storm intensity distribution
\end{itemize}

\mkmframework

\subsection{UMAP Core Theory}

\subsubsection{Local Metric Approximation}
For each high-dimensional point $x_i$ in the weather state space, UMAP constructs local
neighbourhood relationships. The local connectivity radius is:
\begin{equation}
    \rho_i = \min \{ d(x_i, x_{ij}) \mid 1 \leq j \leq k, \; d(x_i, x_{ij}) > 0 \}
\end{equation}

The bandwidth parameter $\sigma_i$ is calibrated to ensure consistent local connectivity:
\begin{equation}
    \sum_{j=1}^k \exp\left(-\frac{\max(0,\, d(x_i, x_{ij}) - \rho_i)}{\sigma_i}\right)
    = \log_2(k)
\end{equation}

This adapts to varying densities in atmospheric data, where specific weather patterns may
be densely clustered while others are sparsely distributed.

\subsubsection{Fuzzy Topological Representation}
UMAP constructs a weighted $k$-nearest neighbour graph with edge weights representing
fuzzy set membership:
\begin{equation}
    w_{ij} = \exp\left(-\frac{\max(0,\, d(x_i, x_j) - \rho_i)}{\sigma_i}\right)
\end{equation}

The directed adjacency matrix $A$ is symmetrised via a probabilistic $t$-conorm:
\begin{equation}
    B = A + A^\top - A \circ A^\top
\end{equation}
where $\circ$ denotes the Hadamard product. This preserves the topological structure of
weather patterns across varying scales.

\subsubsection{Cross-Entropy Optimisation}
The low-dimensional embedding minimises the cross-entropy between high- and
low-dimensional fuzzy simplicial set representations:
\begin{equation}
    \mathcal{L} = \sum_{(i,j) \in E} \left[
        w_h(i,j) \log \frac{w_h(i,j)}{w_l(i,j)}
        + (1 - w_h(i,j)) \log \frac{1 - w_h(i,j)}{1 - w_l(i,j)}
    \right]
\end{equation}

The low-dimensional similarity uses a Student's $t$-distribution kernel:
\begin{equation}
    w_l(i,j) = \left(1 + a \cdot d(y_i, y_j)^{2b}\right)^{-1}
\end{equation}
with default parameters $a \approx 1.93$, $b \approx 0.79$.

\subsection{Multi-Scale Weather Pattern Integration}

\subsubsection{Scale-Aware Manifold Construction}
Weather systems span synoptic to microscale. We construct a hierarchical manifold
representation:
\begin{equation}
    \mathcal{M}(W(t)) = \bigoplus_{i=1}^{m} \alpha_i \, M_i,
    \qquad M_i = \text{UMAP}(\mathcal{F}_i(W(t)))
\end{equation}
where $\mathcal{F}_i$ applies scale-specific filtering and $\alpha_i$ are
variance-derived importance weights.

\subsubsection{Persistent Homology Filtering}
Topological features are filtered via persistent homology:
\begin{equation}
    PH(M) = \{(b_i, d_i) \mid i = 1, \ldots, n\}
\end{equation}
where $(b_i, d_i)$ are birth--death pairs. Features with persistence above threshold
$\tau$ are retained:
\begin{equation}
    M' = \{x \in M \mid \exists (b_i, d_i) \in PH(M) : (d_i - b_i) > \tau,\;
    x \in \text{Support}(b_i, d_i)\}
\end{equation}

\subsubsection{Weather-to-Flood Mapping}
The reduced representation feeds a physics-constrained neural operator:
\begin{equation}
    F(t + \Delta t) = \Psi(\mathcal{M}(W(t)),\, F(t)),
    \qquad \Psi = \mathcal{D} \circ \mathcal{N} \circ \mathcal{E}
\end{equation}
where $\mathcal{E}$ encodes the combined state, $\mathcal{N}$ integrates forward in time,
and $\mathcal{D}$ decodes subject to physical constraints.

\subsection{Uncertainty Quantification}

\subsubsection{Manifold Distortion Metrics}
Local distortion is quantified via trustworthiness $T(k)$ and continuity $C(k)$:
\begin{equation}
    T(k) = 1 - \frac{2}{nk(2n-3k-1)}\sum_{i=1}^{n}\sum_{j \in U_i^k}(r(i,j) - k)
\end{equation}
\begin{equation}
    C(k) = 1 - \frac{2}{nk(2n-3k-1)}\sum_{i=1}^{n}\sum_{j \in V_i^k}(\hat{r}(i,j) - k)
\end{equation}

\subsubsection{Error Propagation}
Input uncertainties propagate through the manifold mapping via:
\begin{equation}
    \Sigma_{\mathcal{M}} = J_{\mathcal{M}} \, \Sigma_W \, J_{\mathcal{M}}^T
\end{equation}
where $J_{\mathcal{M}}$ is the Jacobian of the UMAP embedding and $\Sigma_W$ is the
weather state covariance matrix.

\subsubsection{Ensemble Approach}
Probabilistic flood prediction uses $N_e$ ensemble members:
\begin{equation}
    P(F(t+\Delta t) \in A) = \frac{1}{N_e}\sum_{i=1}^{N_e}
    \mathbf{1}_{A}(F^{(i)}(t+\Delta t))
\end{equation}

\mkmimplementation

\subsection{Computational Workflow}
\begin{enumerate}[nosep]
    \item Ingest raw HRRR forecast data at 3\,km resolution
    \item Normalise and bias-correct against historical observation--forecast pairs
    \item Apply quality control (missing value detection, anomaly flagging)
    \item Apply scale-aware UMAP dimensionality reduction
    \item Map reduced representations to hydrodynamic initial conditions
    \item Execute ensemble flood simulations (shallow water equations)
    \item Aggregate into probabilistic flood predictions
    \item Project impacts onto property vulnerability assessments
\end{enumerate}

\subsection{Hardware Acceleration}
\begin{itemize}[nosep]
    \item Nearest-neighbour search parallelised via FAISS (GPU)
    \item Stochastic gradient descent for embedding optimisation (GPU)
    \item Batched ensemble processing across multiple GPUs
    \item Approximate 40$\times$ speedup versus CPU implementation
\end{itemize}

\subsection{Topological Preservation}
Topology preservation was quantified using the Wasserstein distance between persistence
diagrams:
\begin{equation}
    W_p(PD_{\text{orig}}, PD_{\text{red}}) =
    \left(\inf_{\gamma} \sum_{x \in PD_{\text{orig}}}
    \|x - \gamma(x)\|_{\infty}^p\right)^{1/p}
\end{equation}

UMAP achieved a 42\% reduction in topological distortion compared to PCA, confirming
superior preservation of watershed boundaries and frontal structures.

\subsection{Historical Backtesting}
The model was backtested against 12 major flood events (2020--2024) in the Thames
catchment. Results are summarised as:
\begin{itemize}[nosep]
    \item 11/12 events: peak water level predicted within $\pm$0.4\,m
    \item 10/12 events: timing of peak within $\pm$3 hours
    \item 1 event (Feb 2022): underpredicted by 0.7\,m due to unprecedented
          antecedent soil saturation not captured in the training data
\end{itemize}

% ===================================================================
% 8. MODEL LIMITATIONS AND KNOWN WEAKNESSES
% ===================================================================
\mkmlimitations

\begin{enumerate}
    \item \textbf{Stationarity assumption:} UMAP is trained on historical data and
          assumes the statistical relationship between weather patterns and flood response
          is stationary. Climate change may invalidate this over multi-decade horizons.

    \item \textbf{Static embedding:} Standard UMAP does not model temporal evolution.
          Sequential embeddings are used as a workaround but may miss rapid regime
          transitions (e.g., sudden convective initiation).

    \item \textbf{Conservation law compliance:} The dimensionality reduction does not
          explicitly enforce mass/energy/momentum conservation. Post-processing validation
          is applied but is not guaranteed to detect all violations.

    \item \textbf{Training data coverage:} Extreme events beyond the historical training
          set (2022--2024) may produce embeddings in under-represented regions of the
          manifold, leading to higher prediction uncertainty.

    \item \textbf{Ensemble calibration:} The ensemble spread may underestimate true
          uncertainty for events near the boundary of the training distribution.

    \item \textbf{Spatial resolution gap:} Downscaling from 3\,km atmospheric to 250\,m
          flood resolution introduces interpolation uncertainty not fully captured in the
          error budget.
\end{enumerate}
